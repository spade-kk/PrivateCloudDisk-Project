基础示例


# 基础示例：声明关键字、字符串、数字、时长、变量引用、steps 引用
workflow "basic" {
    metadata { display_name = "基础流程"; version = "1.0.0" }
    variables {
        count = 3
        ratio = 0.618
        scientific = 1.5e3
        template = "hello ${vars.name}"
        timeout = 30s
    }
    trigger { manual {} }
    runtime { timeout = 5m; max_parallel = 2 }
    steps {
        step greet { action log.message { text = vars.label } }
        step loop {
            depends_on greet if vars.count > 0
            foreach item in vars.items { delay 100ms }
        }
    }
}

高级示例（V1.2 新增语法）


# 覆盖 switch / for / parallel / notify / validate / 管道 / 插值 / 注解
import "common.flow" as common
namespace com.example.workflows
tag "finance"
audit { level = "high"; description = "关键对账流程" }

workflow "advanced" {
    environment { NODE_ENV = "production" }
    trigger { interval = 5m }
    steps {
        step branch {
            switch vars.status {
                case "active" => { delay 1s }
                default => { assert vars.ok == true }
            }
        }
        step loop {
            for i in range(0, vars.max) {
                if vars.skip != true { break }
                continue
            }
        }
        step fan { parallel(max_concurrency = 3) { step a { action svc.a {} } } }
        step guard {
            expect { vars.result > 0 }
            on_error { notify { channel = "sms"; message = "失败" } }
            action data.transform { value = vars.rows | filter(size > 100) | map(name) | reduce(join) }
        }
    }
}

高级示例 full_workflow.flow


# 覆盖目标：V1.1 组合流程：变量、DAG、if、foreach、parallel、try/catch/finally、wait、assert、retry。
# 预期：Compiler 保留全部控制流语义；Runtime 只静态调度根节点，由控制节点动态执行子图。
workflow "full_workflow" {
    metadata { display_name = "CloudFlow 全语法覆盖" version = "1.1" tags = ["coverage", "v1"] }
    trigger { manual {} }
    runtime { timeout = 10m max_parallel = 2 retry_policy { max_attempts = 2 strategy = "exponential" } }
    variables {
        source = input.string(required = true)
        files: array = ["one", "two"]
        threshold: number = 1
        mode = "standard"
    }
    
    step collect { action builtin.file.list { node = vars.source } output files_result }

    assert { size(vars.files) >= vars.threshold }

    if { steps.collect.output.count > 0 } {
        foreach item in vars.files {
            step process { action builtin.file.copy { file = item } retry { max_attempts = 2 strategy = "fixed" } }
        }
    } else {
        step notify_empty { action builtin.notification.send { title = "empty" } }
    }

    parallel {
        step index { action builtin.file.list {} }
        step announce { action builtin.notification.send { title = vars.mode == "standard" ? "started" : "custom" } }
    }

    try {
        step publish { action plugin.reporter.publish { source = steps.collect.output } }
    } catch error {
        step report_failure { action builtin.notification.send { error = error } }
    } finally {
        step audit { action builtin.notification.send { title = "audit" } }
    }

    wait approval { timeout = 24h }

    step finalize { action builtin.notification.send { title = "done" } }
}

高级示例 expression_complex.flow


# 覆盖目标：算术、比较、逻辑、函数、成员访问和三元表达式。
# 预期：所有复杂值使用结构化 $expr 而不是字符串。
workflow "expression_complex" {
    variables {
        x: number = 8
        y: number = 2
        files: array = ["a", "b"]
        options: object = {"region": "cn", "enabled": true}
    }
    step collect { action builtin.file.list {} output result }
    step evaluate {
        depends_on collect
        condition {
            vars.x > 5 && !(vars.y > 10) ||
            size(vars.files) > 0 && contains(vars.files, "a") &&
            starts_with(vars.options.region, "c") && ends_with("CloudFlow", "Flow") &&
            vars.x / vars.y >= 4 && vars.x % vars.y == 0
        }
        action builtin.notification.send {
            score = vars.x * 2 + vars.y
            selected = vars.x > 5 ? "high" : "low"
            file_count = size(vars.files)
            upstream = steps.collect.output.files
            typed_object = {"region": vars.options.region, "count": size(vars.files)}
        }
    }
}

案例示范 weekly_sales_report.flow


workflow "weekly_sales_report" {
    metadata {
        display_name = "销售周报"
        description = "每周一自动生成销售周报"
        version = "1.0"
    }
    trigger {
        schedule {
            cron = "0 8 * * 1"
            timezone = "Asia/Shanghai"
        }
    }
    runtime {
        timeout = 30m
        max_parallel = 4
        retry_policy {
            max_attempts = 3
            strategy = "exponential"
        }
    }
    variables {
        sales_node_id = input.string(required = true)
        template_file_id = input.string(required = true)
        report_node_id = input.string(required = true)
    }
    step collect_files {
        name = "收集销售文件"
        action file.list {
            node = vars.sales_node_id
            filter { extension = "xlsx" }
        }
        output excel_files
    }
    step aggregate_data {
        name = "销售数据统计"
        depends_on collect_files
        action data.aggregate_excel {
            input { files = collect_files.output }
            group_by = "region"
            metrics { sum("sales") average("profit") }
        }
        output report_data
    }
    step generate_report {
        name = "生成销售报告"
        depends_on aggregate_data
        condition { aggregate_data.output.row_count > 0 }
        action plugin {
            id = "8ae47c8d-41c5-4b9d-87e7-2f93b74d34d7"
            function = "generate_report"
            version = "1"
            input { data = aggregate_data.output template = vars.template_file_id }
        }
        retry { max_attempts = 2 backoff = exponential }
        output report_file
    }
    step save_report {
        name = "保存报告"
        depends_on generate_report
        action file.save {
            source = generate_report.output.file_id
            target = vars.report_node_id
        }
    }

    on_failure {
        step notify_failure {
            action notification.send {
                channel = "user"
                title = "销售周报生成失败"
                message = "失败步骤：${workflow.failed_step}"
            }
        }
    }
}

案例示范 file_approval.flow


#实例 file_approval
workflow "file_approval" {
    metadata {
        display_name = "文件审批"
        description = "文件进入归档目录前发起审批并等待平台能力返回结果"
        version = "1.0"
    }
    trigger { event { name = "FileUploaded" } }
    variables {
        file_id = input.file(required = true)
        archive_node_id = input.string(required = true)
    }
    step request_approval {
        action approval.request {
            file = vars.file_id
            timeout = 24h
        }
        retry { max_attempts = 2 strategy = "exponential" }
        output approval
    }
    step archive_file {
        depends_on request_approval
        condition { steps.request_approval.output.approved == true }
        action file.move {
            file = vars.file_id
            target = vars.archive_node_id
        }
    }
    on_failure {
        step notify_owner {
            action notification.send {
                channel = "user"
                title = "文件审批流程失败"
                message = "请在工作流执行记录中查看失败节点"
            }
        }
    }
}
